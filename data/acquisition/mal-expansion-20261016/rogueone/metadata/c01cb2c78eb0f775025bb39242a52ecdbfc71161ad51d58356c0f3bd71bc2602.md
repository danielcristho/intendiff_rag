Granule is a fs and build helpers.

## Usage

```js
const babel = require('babel-core');
const { apply, getDirFiles, getFileContent, readFile, moveFile } = require('granule');

apply(
    readFile(__dirname, '../.babelrc'),
    JSON.parse,
    babelConfig => apply(
        getDirFiles(__dirname, '../src'),
        file => Object.assign(file, { content: babel.transform(file.content, babelConfig).code }),
        moveFile(__dirname, '../build')
    )
);
```
